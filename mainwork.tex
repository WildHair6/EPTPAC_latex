\documentclass[tikz,border=8pt]{standalone}
\usepackage{newtxtext,newtxmath}
\usetikzlibrary{arrows.meta, positioning, fit, calc}

\begin{document}
\begin{tikzpicture}[
    block/.style={rectangle, draw, thick, rounded corners=2pt, fill=gray!5,
                  minimum width=4cm, minimum height=1.3cm, align=center, font=\small},
    arrow/.style={-Stealth, thick},
    sig/.style={font=\footnotesize}
]

% ---- Main blocks (vertical flow) ----
\node (planner) [block] {Constrained\\Trajectory Planner};
\node (controller) [block, below=1.5cm of planner] 
    {Nonsingular Prescribed-Time\\Prescribed-Accuracy Controller};
\node (spacecraft) [block, below=1.6cm of controller] 
    {Spacecraft with\\Reaction Wheels};

% ---- Loop labels ----
\node [blue, font=\footnotesize\itshape, anchor=south west] 
    at ([xshift=3pt, yshift=3pt]planner.north west) {Outer loop};
\node [blue, font=\footnotesize\itshape, anchor=south west] 
    at ([xshift=3pt, yshift=3pt]controller.north west) {Inner loop};

% ---- EPTPAC Framework box (contains planner + controller) ----
\node (eptpac) [draw=red!70!black, dashed, thick, rounded corners=4pt,
    inner xsep=18pt, inner ysep=22pt, fit=(planner)(controller)] {};
\node [red!70!black, font=\small\itshape, above=2pt] at (eptpac.north) 
    {EPTPAC Framework};

% ---- Signal arrows ----
% Mission inputs from left
\draw [arrow] ([xshift=-2.5cm]planner.west) -- 
    node[sig, above] {$T_f,\;\boldsymbol{u}_{\max},\;\boldsymbol{h}_{w,\max}$} 
    (planner.west);
\draw [arrow] ([xshift=-2.5cm]controller.west) -- 
    node[sig, above] {$\varepsilon_{\mathrm{mission}},\;\boldsymbol{d}_{\max}$} 
    (controller.west);

% Planner → Controller (vertical inside framework)
\draw [arrow] (planner) -- 
    node[sig, right] 
    {$\boldsymbol{\sigma}_d,\;\boldsymbol{\omega}_d,\;\dot{\boldsymbol{\omega}}_d$} 
    (controller);

% Controller → Spacecraft (vertical, exits framework)
\draw [arrow] (controller) -- node[sig, right] {$\boldsymbol{u}$} (spacecraft);

% Disturbances from left into spacecraft
\draw [arrow] ([xshift=-2cm]spacecraft.west) -- 
    node[sig, above] {$\boldsymbol{d},\;\Delta\boldsymbol{J}$} (spacecraft.west);

% Feedback loop: spacecraft right → up → controller right
\coordinate (fb) at ([xshift=2cm]eptpac.east |- spacecraft);
\draw [arrow] (spacecraft.east) -- 
    node[sig, above] {$\boldsymbol{\sigma},\;\boldsymbol{\omega},\;\boldsymbol{h}_w$}
    (fb) |- (controller.east);

\end{tikzpicture}
\end{document}
